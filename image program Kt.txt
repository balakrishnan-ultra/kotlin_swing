import java.awt.Color
import java.awt.image.BufferedImage
import java.io.File
import javax.imageio.ImageIO

fun main() {
    val width = 200
    val height = 150

    // Create a BufferedImage object
    val image = BufferedImage(width, height, BufferedImage.TYPE_INT_RGB)

    // Set a background color (optional)
    val graphics = image.createGraphics()
    graphics.color = Color.WHITE
    graphics.fillRect(0, 0, width, height)
    graphics.dispose() // Release resources

    // Save the image to a file
    val outputFile = File("blank_image.png")
    ImageIO.write(image, "png", outputFile)

    println("Blank image saved to ${outputFile.absolutePath}")
}

import java.awt.Color
import java.awt.image.BufferedImage
import java.io.File
import javax.imageio.ImageIO

fun main() {
    val inputFile = File("input.jpg") // Replace with your image file
    try {
        val image = ImageIO.read(inputFile)
        if (image == null) {
            println("Could not read the image file.")
            return
        }

        val width = image.width
        val height = image.height

        // Example: Invert the colors of the image
        for (y in 0 until height) {
            for (x in 0 until width) {
                val rgb = image.getRGB(x, y)
                val alpha = (rgb shr 24) and 0xFF
                val red = 255 - ((rgb shr 16) and 0xFF)
                val green = 255 - ((rgb shr 8) and 0xFF)
                val blue = 255 - (rgb and 0xFF)

                val newRgb = (alpha shl 24) or (red shl 16) or (green shl 8) or blue
                image.setRGB(x, y, newRgb)
            }
        }

        val outputFile = File("inverted_image.png")
        ImageIO.write(image, "png", outputFile)
        println("Inverted image saved to ${outputFile.absolutePath}")

    } catch (e: Exception) {
        println("Error processing image: ${e.message}")
    }
}
